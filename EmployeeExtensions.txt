package com.example.employeeinfo_2415141122115

import java.text.NumberFormat
import java.util.Locale
fun Employee.getUpperCaseName(): String {
    return hoTen.uppercase()
}
fun Employee.getFormattedLuong(): String {
    val formatter = NumberFormat.getCurrencyInstance(Locale("vi", "VN"))
    return formatter.format(luong)
}
fun Employee.getThamNienLevel(): String {
    return when {
        thamNien >= 5 -> "Nhân viên lâu năm (Senior)"
        thamNien >= 2 -> "Nhân viên có kinh nghiệm (Mid-level)"
        else -> "Nhân viên mới (Junior)"
    }
}
fun Employee.getDisplayInfo(): String {
    return """
        Mã nhân viên: $maNV
        Họ tên: ${getUpperCaseName()}
        Phòng ban: $phongBan
        Giới tính: $gioiTinh
        Tuổi: $tuoi
        Lương: ${getFormattedLuong()}
        Thâm niên: $thamNien năm (${getThamNienLevel()})
    """.trimIndent()
}
